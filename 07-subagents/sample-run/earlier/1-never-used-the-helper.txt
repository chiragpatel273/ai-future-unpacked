def percent_off(value, items):
    """Take value% off the non‑sale items in the cart.

    Sale items are already discounted, so coupons should not stack on them,
    but they also should not prevent discounts on other items. This function
    now only considers items where ``on_sale`` is ``False`` when calculating
    the eligible amount.
    """
    # Sum only items that are not on sale.
    eligible = sum(i.price * i.qty for i in items if not i.on_sale)
    return round(eligible * value / 100, 2)


def amount_off(value, items):
    """Take a fixed amount off, but never more than the cart is worth."""
    total = sum(i.price * i.qty for i in items)
    return min(value, total)


RULES = {"percent_off": percent_off, "amount_off": amount_off}


def discount_for(coupon, items):
    return RULES[coupon["rule"]](coupon["value"], items)
